async function updateTotal(){ const el=document.getElementById('total-counter'); const visitedKey='abacus_visited'; const cacheKey='abacus_cache'; // Si ya visitó hoy, no vuelvas a hacer hit, usa cache const lastVisit=localStorage.getItem(visitedKey); const today=new Date().toDateString(); let url='https://abacus.jasoncameron.dev/hit/spikeblack009-blog/total'; if(lastVisit===today){ url='https://abacus.jasoncameron.dev/get/spikeblack009-blog/total'; // solo lee, no suma const cached=localStorage.getItem(cacheKey); if(cached){ const d=JSON.parse(cached); if(Date.now()-d.t < 3600000){ // 1 hora cache el.textContent=d.value; return; } } } try{ const r=await fetch(url,{cache:'no-store'}); const d=await r.json(); if(d.value){ el.textContent=d.value; localStorage.setItem(cacheKey,JSON.stringify({t:Date.now(),value:d.value})); localStorage.setItem(visitedKey,today); } }catch(e){ const cached=localStorage.getItem(cacheKey); el.textContent=cached?JSON.parse(cached).value:'1'; } }
